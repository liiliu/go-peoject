# 数据库操作指南 (GORM)

## 📋 概述

项目使用 **GORM** 作为 ORM 框架，提供了丰富的数据库操作功能。

import "your_project/library/database"

## 📝 基础操作

### 1. 创建记录

```go
// 创建单条
user := &model.User{
    Username: "张三",
    Email:    "zhangsan@example.com",
    Password: "123456",
}
result := database.DB.Create(user)

// 检查错误
if result.Error != nil {
    logger.Error("db", "创建失败: %v", result.Error)
}

// 获取自增 ID
logger.Info("db", "新用户 ID: %d", user.ID)

// 批量创建
users := []model.User{
    {Username: "user1", Email: "user1@example.com"},
    {Username: "user2", Email: "user2@example.com"},
}
database.DB.Create(&users)
```

### 2. 查询记录

```go
// 查询单条
var user model.User
database.DB.First(&user, 1)  // 根据主键查询
database.DB.First(&user, "email = ?", "test@example.com")

// 查询所有
var users []model.User
database.DB.Find(&users)

// 条件查询
database.DB.Where("status = ?", 1).Find(&users)
database.DB.Where("age > ?", 18).Find(&users)
database.DB.Where("username LIKE ?", "%张%").Find(&users)

// 多条件
database.DB.Where("status = ? AND age > ?", 1, 18).Find(&users)

// 排序
database.DB.Order("created_at desc").Find(&users)

// 限制数量
database.DB.Limit(10).Find(&users)

// 分页
page := 1
size := 10
database.DB.Limit(size).Offset((page - 1) * size).Find(&users)
```

### 3. 更新记录

```go
// 更新单个字段
database.DB.Model(&user).Update("status", 1)

// 更新多个字段
database.DB.Model(&user).Updates(map[string]interface{}{
    "username": "新名字",
    "email":    "new@example.com",
})

// 更新结构体（只更新非零值）
database.DB.Model(&user).Updates(model.User{
    Username: "新名字",
    Status:   1,
})

// 更新所有字段（包括零值）
database.DB.Model(&user).Select("*").Updates(model.User{
    Username: "新名字",
    Status:   0,  // 即使是 0 也会更新
})

// 批量更新
database.DB.Model(&model.User{}).
    Where("status = ?", 0).
    Update("status", 1)
```

### 4. 删除记录

```go
// 软删除（DeletedAt 字段）
database.DB.Delete(&user, 1)

// 永久删除
database.DB.Unscoped().Delete(&user, 1)

// 批量删除
database.DB.Where("status = ?", 0).Delete(&model.User{})
```

## 🔍 高级查询

### 1. Select 指定字段

```go
// 只查询指定字段
var users []model.User
database.DB.Select("id, username, email").Find(&users)

// 排除某些字段
database.DB.Omit("password").Find(&users)
```

### 2. Count 统计

```go
var count int64
database.DB.Model(&model.User{}).Where("status = ?", 1).Count(&count)
logger.Info("db", "活跃用户数: %d", count)
```

### 3. Pluck 获取单列

```go
var usernames []string
database.DB.Model(&model.User{}).Pluck("username", &usernames)

var ids []uint
database.DB.Model(&model.User{}).Pluck("id", &ids)
```

### 4. Scan 扫描到自定义结构

```go
type Result struct {
    Username string
    Email    string
}

var results []Result
database.DB.Model(&model.User{}).
    Select("username, email").
    Where("status = ?", 1).
    Scan(&results)
```

### 5. 原生 SQL

```go
// 查询
var users []model.User
database.DB.Raw("SELECT * FROM users WHERE status = ?", 1).Scan(&users)

// 执行
database.DB.Exec("UPDATE users SET status = ? WHERE age < ?", 0, 18)

// 统计
var count int64
database.DB.Raw("SELECT COUNT(*) FROM users WHERE status = ?", 1).Scan(&count)
```

## 🔗 关联查询

### 1. 一对多

```go
// 定义模型
type User struct {
    ID     uint
    Name   string
    Orders []Order  // 一对多
}

type Order struct {
    ID     uint
    UserID uint
    Amount float64
}

// 预加载
var users []User
database.DB.Preload("Orders").Find(&users)

// 条件预加载
database.DB.Preload("Orders", "amount > ?", 100).Find(&users)
```

### 2. 一对一

```go
type User struct {
    ID      uint
    Name    string
    Profile Profile  // 一对一
}

type Profile struct {
    ID     uint
    UserID uint
    Avatar string
}

// 预加载
var user User
database.DB.Preload("Profile").First(&user, 1)
```

### 3. 多对多

```go
type User struct {
    ID    uint
    Name  string
    Roles []Role `gorm:"many2many:user_roles;"`
}

type Role struct {
    ID   uint
    Name string
}

// 预加载
var users []User
database.DB.Preload("Roles").Find(&users)
```

### 4. Join 查询

```go
type Result struct {
    UserID   uint
    Username string
    OrderNo  string
    Amount   float64
}

var results []Result
database.DB.Table("users").
    Select("users.id as user_id, users.username, orders.order_no, orders.amount").
    Joins("left join orders on orders.user_id = users.id").
    Where("orders.amount > ?", 100).
    Scan(&results)
```

## 🔄 事务

### 1. 手动事务

```go
func CreateUserWithProfile(user *model.User, profile *model.Profile) error {
    // 开始事务
    tx := database.DB.Begin()
    defer func() {
        if r := recover(); r != nil {
            tx.Rollback()
        }
    }()
    
    // 创建用户
    if err := tx.Create(user).Error; err != nil {
        tx.Rollback()
        return err
    }
    
    // 创建资料
    profile.UserID = user.ID
    if err := tx.Create(profile).Error; err != nil {
        tx.Rollback()
        return err
    }
    
    // 提交事务
    return tx.Commit().Error
}
```

### 2. 自动事务

```go
func CreateUserWithProfile(user *model.User, profile *model.Profile) error {
    return database.DB.Transaction(func(tx *gorm.DB) error {
        // 创建用户
        if err := tx.Create(user).Error; err != nil {
            return err  // 自动回滚
        }
        
        // 创建资料
        profile.UserID = user.ID
        if err := tx.Create(profile).Error; err != nil {
            return err  // 自动回滚
        }
        
        // 返回 nil 自动提交
        return nil
    })
}
```

## 🎯 实战示例

### 示例 1：用户分页查询

```go
func GetUserList(page, size int, keyword string) ([]model.User, int64, error) {
    var users []model.User
    var total int64
    
    query := database.DB.Model(&model.User{})
    
    // 搜索条件
    if keyword != "" {
        query = query.Where("username LIKE ? OR email LIKE ?", 
            "%"+keyword+"%", "%"+keyword+"%")
    }
    
    // 统计总数
    query.Count(&total)
    
    // 分页查询
    err := query.
        Order("created_at desc").
        Limit(size).
        Offset((page - 1) * size).
        Find(&users).Error
    
    return users, total, err
}
```

### 示例 2：订单统计

```go
type OrderStats struct {
    TotalCount  int64
    TotalAmount float64
    AvgAmount   float64
}

func GetOrderStats(userID uint) (*OrderStats, error) {
    var stats OrderStats
    
    err := database.DB.Model(&model.Order{}).
        Where("user_id = ?", userID).
        Select("COUNT(*) as total_count, SUM(amount) as total_amount, AVG(amount) as avg_amount").
        Scan(&stats).Error
    
    return &stats, err
}
```

### 示例 3：批量插入优化

```go
func BatchCreateUsers(users []model.User) error {
    // 分批插入，每次 1000 条
    batchSize := 1000
    
    for i := 0; i < len(users); i += batchSize {
        end := i + batchSize
        if end > len(users) {
            end = len(users)
        }
        
        batch := users[i:end]
        if err := database.DB.CreateInBatches(batch, batchSize).Error; err != nil {
            return err
        }
        
        logger.Info("db", "已插入 %d/%d", end, len(users))
    }
    
    return nil
}
```

## 🔧 性能优化

### 1. 使用索引

```go
type User struct {
    ID       uint   `gorm:"primarykey"`
    Email    string `gorm:"size:100;uniqueIndex"` // 唯一索引
    Username string `gorm:"size:50;index"`        // 普通索引
    Status   int    `gorm:"index"`
}

// 复合索引
type Order struct {
    ID        uint `gorm:"primarykey"`
    UserID    uint `gorm:"index:idx_user_status"` // 复合索引
    Status    int  `gorm:"index:idx_user_status"`
    CreatedAt time.Time
}
```

### 2. 批量查询

```go
// ❌ N+1 问题
var users []model.User
database.DB.Find(&users)
for _, user := range users {
    var orders []model.Order
    database.DB.Where("user_id = ?", user.ID).Find(&orders)  // N次查询
}

// ✅ 使用 Preload
var users []model.User
database.DB.Preload("Orders").Find(&users)  // 2次查询
```

### 3. Select 必要字段

```go
// ❌ 查询所有字段
database.DB.Find(&users)

// ✅ 只查询需要的字段
database.DB.Select("id, username, email").Find(&users)
```

### 4. 使用连接池

```go
// 在 library/database/database.go 中配置
sqlDB, _ := db.DB()
sqlDB.SetMaxIdleConns(10)           // 最大空闲连接数
sqlDB.SetMaxOpenConns(100)          // 最大打开连接数
sqlDB.SetConnMaxLifetime(time.Hour) // 连接最大生命周期
```

## ⚠️ 注意事项

### 1. 避免 SQL 注入

```go
// ❌ 危险！
username := c.Query("username")
database.DB.Where(fmt.Sprintf("username = '%s'", username)).Find(&users)

// ✅ 使用参数绑定
database.DB.Where("username = ?", username).Find(&users)
```

### 2. 错误处理

```go
// ❌ 忽略错误
database.DB.Create(&user)

// ✅ 检查错误
if err := database.DB.Create(&user).Error; err != nil {
    logger.Error("db", "创建失败: %v", err)
    return err
}
```

### 3. 事务回滚

```go
// ❌ 没有 defer
tx := database.DB.Begin()
if err := tx.Create(&user).Error; err != nil {
    tx.Rollback()
    return err
}

// ✅ 使用 defer 确保回滚
tx := database.DB.Begin()
defer func() {
    if r := recover(); r != nil {
        tx.Rollback()
    }
}()
```
