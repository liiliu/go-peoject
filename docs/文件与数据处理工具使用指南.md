# 文件与数据处理工具使用指南

## 📋 概述

项目提供了三个强大的数据处理工具库：
- **Excel** - Excel 文件读写、样式设置、数据导出
- **CSV** - CSV 文件读写、数据转换
- **File** - 文件下载、压缩、哈希、批量操作

---

## 📊 Excel 工具 (`library/excel`)

### 安装依赖

```bash
go get github.com/xuri/excelize/v2
```

### 基础用法

#### 1. 读取 Excel

```go
import "your_project/library/excel"

// 读取整个 Sheet
rows, err := excel.ReadSheet("users.xlsx", "Sheet1")
for _, row := range rows {
    fmt.Println(row)  // ["张三", "25", "北京"]
}

// 读取第一个 Sheet
rows, err := excel.ReadFirstSheet("users.xlsx")

// 读取为 Map（第一行作为表头）
data, err := excel.ReadSheetToMap("users.xlsx", "Sheet1")
for _, record := range data {
    fmt.Printf("姓名: %s, 年龄: %s\n", record["姓名"], record["年龄"])
}

// 读取单个单元格
value, err := excel.GetCellValue("users.xlsx", "Sheet1", "A1")
```

#### 2. 写入 Excel

```go
// 写入二维数组
data := [][]interface{}{
    {"姓名", "年龄", "城市"},
    {"张三", 25, "北京"},
    {"李四", 30, "上海"},
}
err := excel.WriteData("output.xlsx", "用户数据", data)

// 写入 Map 数据（自动生成表头）
users := []map[string]interface{}{
    {"name": "张三", "age": 25, "city": "北京"},
    {"name": "李四", "age": 30, "city": "上海"},
}
err := excel.WriteMapData("users.xlsx", "Sheet1", users)

// 追加行到现有文件
row := []interface{}{"王五", 28, "广州"}
err := excel.AppendRow("users.xlsx", "Sheet1", row)
```

#### 3. 带样式导出

```go
headers := []string{"姓名", "年龄", "城市", "职位"}
data := [][]interface{}{
    {"张三", 25, "北京", "工程师"},
    {"李四", 30, "上海", "经理"},
    {"王五", 28, "广州", "设计师"},
}

// 自动添加：表头样式、列宽、筛选、冻结首行
err := excel.ExportData("员工表.xlsx", "员工信息", headers, data)
```

效果：
- ✅ 蓝色表头 + 白色字体
- ✅ 居中对齐
- ✅ 自动筛选
- ✅ 冻结首行
- ✅ 自动列宽

#### 4. 高级样式

```go
f := excel.NewExcel()

// 设置单元格样式
style := &excelize.Style{
    Font: &excelize.Font{
        Bold: true,
        Size: 14,
        Color: "FF0000",
    },
    Fill: excelize.Fill{
        Type:    "pattern",
        Color:   []string{"FFFF00"},
        Pattern: 1,
    },
}
excel.SetCellStyle(f, "Sheet1", "A1", "C1", style)

// 设置列宽
excel.SetColumnWidth(f, "Sheet1", "A", "C", 20)

// 自动筛选
excel.AutoFilterTable(f, "Sheet1", "A1", "C10")

f.SaveAs("styled.xlsx")
```

#### 5. 合并 Excel 文件

```go
files := []string{
    "2024-01.xlsx",
    "2024-02.xlsx",
    "2024-03.xlsx",
}
err := excel.MergeExcelFiles("全年汇总.xlsx", files)
```

### 实战示例：导出用户数据

```go
package service

import "your_project/library/excel"

func ExportUsers() error {
    // 从数据库查询用户
    var users []model.User
    database.DB.Find(&users)
    
    // 准备数据
    headers := []string{"ID", "用户名", "邮箱", "手机号", "注册时间"}
    data := make([][]interface{}, 0, len(users))
    
    for _, user := range users {
        row := []interface{}{
            user.ID,
            user.Username,
            user.Email,
            user.Mobile,
            user.CreatedAt.Format("2006-01-02 15:04:05"),
        }
        data = append(data, row)
    }
    
    // 导出
    return excel.ExportData("users.xlsx", "用户列表", headers, data)
}
```

## 📄 CSV 工具 (`library/csv`)

### 基础用法

#### 1. 读取 CSV

```go
import "your_project/library/csv"

// 读取所有数据
data, err := csv.Read("users.csv")
for _, row := range data {
    fmt.Println(row)
}

// 读取为 Map
records, err := csv.ReadToMap("users.csv")
for _, record := range records {
    fmt.Printf("姓名: %s, 年龄: %s\n", record["name"], record["age"])
}

// 使用 Tab 分隔符
data, err := csv.ReadWithCustomDelimiter("data.tsv", '\t')
```

#### 2. 写入 CSV

```go
// 写入数据
data := [][]string{
    {"姓名", "年龄", "城市"},
    {"张三", "25", "北京"},
    {"李四", "30", "上海"},
}
err := csv.Write("output.csv", data)

// 从 Map 写入
headers := []string{"name", "age", "city"}
users := []map[string]string{
    {"name": "张三", "age": "25", "city": "北京"},
    {"name": "李四", "age": "30", "city": "上海"},
}
err := csv.WriteFromMap("users.csv", headers, users)

// 追加数据
newData := [][]string{
    {"王五", "28", "广州"},
}
err := csv.Append("users.csv", newData)
```

func ExportOrdersCSV(c *fiber.Ctx) error {

    filename := "orders.csv"

        return err
    }
    
    return c.Download(filename)
}

## 📁 文件工具 (`library/file`)

### 文件下载

```go
import "your_project/library/file"

// 简单下载
err := file.Download(
    "https://example.com/file.pdf",
    "/path/to/save/file.pdf",
)

// 带进度的下载
err := file.DownloadWithProgress(url, savePath, func(current, total int64) {
    progress := float64(current) / float64(total) * 100
    fmt.Printf("\r下载进度: %.2f%%", progress)
})

// 下载到内存
data, err := file.DownloadToBytes("https://example.com/image.jpg")
```

### 文件操作

```go
// 判断文件是否存在
if file.Exists("/path/to/file.txt") {
    fmt.Println("文件存在")
}

// 判断是否为文件/目录
file.IsFile("/path/to/file.txt")   // true
file.IsDir("/path/to/directory")   // true

// 获取文件大小
size, err := file.FileSize("/path/to/file.txt")
fmt.Println(file.FileSizeFormat(size))  // "1.5 MB"

// 复制文件
err := file.Copy("/path/to/source.txt", "/path/to/dest.txt")

// 移动文件
err := file.Move("/path/to/old.txt", "/path/to/new.txt")

// 删除文件
err := file.Delete("/path/to/file.txt")

// 读写文件
data, err := file.ReadFile("/path/to/file.txt")
err := file.WriteFile("/path/to/file.txt", []byte("content"))
err := file.AppendFile("/path/to/file.txt", []byte("more content"))
```

### 文件类型检测

```go
// 获取 MIME 类型
mimeType, err := file.GetMimeType("/path/to/file.pdf")
// "application/pdf"

// 获取扩展名
ext := file.GetExtension("/path/to/file.txt")         // ".txt"
ext := file.GetExtensionWithoutDot("/path/to/file.txt") // "txt"

// 获取文件名
name := file.GetFileName("/path/to/file.txt")        // "file"
name := file.GetFileNameWithExt("/path/to/file.txt") // "file.txt"

// 获取目录
dir := file.GetDir("/path/to/file.txt")  // "/path/to"
```

### 压缩与解压

```go
// 压缩文件
files := []string{"file1.txt", "file2.txt", "file3.txt"}
err := file.ZipFiles("archive.zip", files)

// 解压文件
err := file.UnzipFile("archive.zip", "/path/to/extract")
```

### 文件哈希

```go
// 计算 MD5
md5Hash, err := file.MD5File("/path/to/file.txt")
fmt.Println("MD5:", md5Hash)
```

### 批量操作

```go
// 列出所有文件
files, err := file.ListFiles("/path/to/dir")
for _, f := range files {
    fmt.Println(f)
}

// 列出指定类型文件
pdfFiles, err := file.ListFilesByExt("/path/to/dir", ".pdf")

// 删除指定类型文件
err := file.DeleteFilesByExt("/path/to/dir", ".tmp")
```

## 💡 综合实战示例

### 示例 1：用户数据导入

```go
func ImportUsers(c *fiber.Ctx) error {
    // 接收上传的 Excel 文件
    fileHeader, err := c.FormFile("file")
    if err != nil {
        return err
    }
    
    // 保存临时文件
    tempPath := "/tmp/" + fileHeader.Filename
    if err := c.SaveFile(fileHeader, tempPath); err != nil {
        return err
    }
    defer file.Delete(tempPath)
    
    // 读取 Excel 数据
    data, err := excel.ReadSheetToMap(tempPath, "Sheet1")
    if err != nil {
        return err
    }
    
    // 批量导入
    for _, record := range data {
        user := &model.User{
            Username: record["用户名"],
            Email:    record["邮箱"],
            Mobile:   record["手机号"],
        }
        database.DB.Create(user)
    }
    
    return c.JSON(fiber.Map{
        "code": 200,
        "msg":  fmt.Sprintf("成功导入 %d 条数据", len(data)),
    })
}
```

### 示例 2：文件上传与处理

```go
func UploadAndProcess(c *fiber.Ctx) error {
    fileHeader, _ := c.FormFile("file")
    
    // 检查文件类型
    ext := file.GetExtension(fileHeader.Filename)
    allowed := []string{".xlsx", ".xls", ".csv"}
    if !util.InStringSlice(ext, allowed) {
        return errors.New("不支持的文件类型")
    }
    
    // 生成新文件名
    newName := fmt.Sprintf("%s%s", util.RandomString(16), ext)
    savePath := filepath.Join("uploads", newName)
    
    // 保存文件
    if err := c.SaveFile(fileHeader, savePath); err != nil {
        return err
    }
    
    // 计算文件 MD5（用于去重）
    md5Hash, _ := file.MD5File(savePath)
    
    // 保存到数据库
    attachment := &model.Attachment{
        Name:     fileHeader.Filename,
        Path:     savePath,
        Size:     fileHeader.Size,
        MD5:      md5Hash,
        MimeType: fileHeader.Header.Get("Content-Type"),
    }
    database.DB.Create(attachment)
    
    return c.JSON(fiber.Map{
        "code": 200,
        "data": attachment,
    })
}
```

### 示例 3：数据导出（支持多格式）

```go
func ExportData(c *fiber.Ctx) error {
    format := c.Query("format", "xlsx") // xlsx, csv
    
    // 查询数据
    var users []model.User
    database.DB.Find(&users)
    
    // 准备数据
    headers := []string{"ID", "用户名", "邮箱", "注册时间"}
    
    if format == "xlsx" {
        // 导出 Excel
        data := make([][]interface{}, 0, len(users))
        for _, user := range users {
            data = append(data, []interface{}{
                user.ID,
                user.Username,
                user.Email,
                user.CreatedAt.Format("2006-01-02"),
            })
        }
        
        filename := "users.xlsx"
        excel.ExportData(filename, "用户列表", headers, data)
        return c.Download(filename)
        
    } else {
        // 导出 CSV
        data := []map[string]string{}
        for _, user := range users {
            data = append(data, map[string]string{
                "ID":      fmt.Sprintf("%d", user.ID),
                "用户名":    user.Username,
                "邮箱":     user.Email,
                "注册时间":   user.CreatedAt.Format("2006-01-02"),
            })
        }
        
        filename := "users.csv"
        csv.WriteFromMap(filename, headers, data)
        return c.Download(filename)
    }
}
```

func DownloadAndConvert(url string) error {
    // 下载远程文件
    tempFile := "/tmp/download.xlsx"
    if err := file.Download(url, tempFile); err != nil {
        return err
    }
    defer file.Delete(tempFile)

    data, err := excel.ReadFirstSheet(tempFile)
    if err != nil {
        return err
    }
    
    // 转换为 CSV
    csvFile := "output.csv"
    return csv.Write(csvFile, data)

## ⚠️ 注意事项

### Excel
- 大文件（>10MB）读取较慢，考虑分批处理
- 样式设置会增加文件大小
- 保存时记得 `defer f.Close()`

### CSV
- 默认使用逗号分隔符
- 字段包含逗号时会自动转义
- 大文件建议逐行读取

### File
- 下载大文件建议使用带进度的方法
- 压缩时注意文件路径
- 批量操作前做好备份
